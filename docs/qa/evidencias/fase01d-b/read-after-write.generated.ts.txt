function n0_click(ctx: SemanticRuntime, state: SemanticState): void {
    ctx.visit({
        "nodeId": "click"
    });
    ctx.recordControl({
        "nodeId": "click"
    });
    n7_assign(ctx, state);
}
function n1_counter(ctx: SemanticRuntime, state: SemanticState): number {
    ctx.visit({
        "nodeId": "counter"
    });
    const value: number = ctx.readVariable(state, "count") as number;
    ctx.validateValue(value, "number", {
        "nodeId": "counter"
    });
    ctx.recordValue({
        "nodeId": "counter"
    }, value);
    return value;
}
